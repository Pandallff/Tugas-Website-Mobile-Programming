@Composable
fun BMICalculatorApp() {
    var weightInput by remember { mutableStateOf("") }
    var heightInput by remember { mutableStateOf("") }
    var bmiResult by remember { mutableStateOf(null) }
    var bmiCategory by remember { mutableStateOf("") }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(24.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Top
    ) {
        Text(
            text = "BMI Calculator",
            style = MaterialTheme.typography.headlineMedium
        )

        Spacer(modifier = Modifier.height(24.dp))

        OutlinedTextField(
            value = weightInput,
            onValueChange = { weightInput = it },
            label = { Text("Berat Badan (kg)") },
            keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
            singleLine = true,
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(modifier = Modifier.height(16.dp))

        OutlinedTextField(
            value = heightInput,
            onValueChange = { heightInput = it },
            label = { Text("Tinggi Badan (cm)") },
            keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
            singleLine = true,
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(modifier = Modifier.height(24.dp))

        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            Button(
                onClick = {
                    val weight = weightInput.replace(',', '.').toFloatOrNull()
                    val heightCm = heightInput.replace(',', '.').toFloatOrNull()

                    if (weight != null && heightCm != null && heightCm > 0) {
                        val heightM = heightCm / 100
                        val bmi = weight / (heightM * heightM)
                        bmiResult = bmi

                        bmiCategory = when {
                            bmi < 18.5f -> "Kekurangan Berat Badan"
                            bmi in 18.5f..24.9f -> "Normal"
                            bmi in 25.0f..29.9f -> "Kelebihan Berat Badan"
                            else -> "Obesitas"
                        }
                    }
                },
                modifier = Modifier.weight(1f)
            ) {
                Text("Hitung BMI")
            }

            Spacer(modifier = Modifier.width(12.dp))

            OutlinedButton(
                onClick = {
                    weightInput = ""
                    heightInput = ""
                    bmiResult = null
                    bmiCategory = ""
                },
                modifier = Modifier.weight(1f)
            ) {
                Text("Reset")
            }
        }

        Spacer(modifier = Modifier.height(8.dp))

        TextButton(
            onClick = {
                bmiCategory = "Acuan: <18.5 Kurang, 18.5-24.9 Normal, >=25 Lebih"
            }
        ) {
            Text("Lihat Kategori")
        }

        Spacer(modifier = Modifier.height(32.dp))

        bmiResult?.let { bmi ->
            Text(
                text = "Hasil BMI",
                style = MaterialTheme.typography.titleMedium
            )
            Text(
                text = String.format(Locale.US, "%.1f", bmi),
                style = MaterialTheme.typography.displayMedium,
                color = MaterialTheme.colorScheme.primary
            )
            Text(
                text = bmiCategory,
                style = MaterialTheme.typography.titleLarge
            )
        } ?: run {
            if (bmiCategory.isNotEmpty()) {
                Text(
                    text = bmiCategory,
                    style = MaterialTheme.typography.bodyLarge
                )
            }
        }
    }
}

package com.example.quiz1.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.example.quiz1.model.Student
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.stateIn

class StudentViewModel : ViewModel() {

    private val initialStudents = listOf(
        Student(id = "1", nim = "2301001", name = "Budi Santoso", programStudi = "Informatika"),
        Student(id = "2", nim = "2301002", name = "Siti Aminah", programStudi = "Sistem Informasi"),
        Student(id = "3", nim = "2301003", name = "Andi Wijaya", programStudi = "Teknik Komputer"),
    )

    private val _students = MutableStateFlow(initialStudents)
    val students: StateFlow<List<Student>> = _students.asStateFlow()

    private val _searchQuery = MutableStateFlow("")
    val searchQuery: StateFlow<String> = _searchQuery.asStateFlow()

    val filteredStudents: StateFlow<List<Student>> = combine(_students, _searchQuery) { list, query ->
        if (query.isBlank()) {
            list
        } else {
            val q = query.lowercase().trim()
            list.filter { student ->
                student.name.lowercase().contains(q) ||
                        student.nim.lowercase().contains(q) ||
                        student.programStudi.lowercase().contains(q)
            }
        }
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5000),
        initialValue = initialStudents,
    )

    fun onSearchQueryChange(query: String) {
        _searchQuery.value = query
    }

    fun addStudent(nim: String, name: String, programStudi: String): Result<Unit> {
        val trimmedNim = nim.trim()
        val trimmedName = name.trim()

        if (trimmedNim.isEmpty()) {
            return Result.failure(IllegalArgumentException("NIM tidak boleh kosong"))
        }
        if (trimmedName.isEmpty()) {
            return Result.failure(IllegalArgumentException("Nama tidak boleh kosong"))
        }
        if (programStudi.isBlank() || (programStudi == "Pilih program studi")) {
            return Result.failure(IllegalArgumentException("Silakan pilih program studi"))
        }

        if (_students.value.any { it.nim.equals(trimmedNim, ignoreCase = true) }) {
            return Result.failure(IllegalArgumentException("NIM $trimmedNim sudah terdaftar"))
        }

        val newStudent = Student(
            nim = trimmedNim,
            name = trimmedName,
            programStudi = programStudi,
        )
        _students.value += newStudent
        return Result.success(Unit)
    }

    fun updateStudent(id: String, nim: String, name: String, programStudi: String): Result<Unit> {
        val trimmedNim = nim.trim()
        val trimmedName = name.trim()

        if (trimmedNim.isEmpty()) {
            return Result.failure(IllegalArgumentException("NIM tidak boleh kosong"))
        }
        if (trimmedName.isEmpty()) {
            return Result.failure(IllegalArgumentException("Nama tidak boleh kosong"))
        }
        if (programStudi.isBlank() || (programStudi == "Pilih program studi")) {
            return Result.failure(IllegalArgumentException("Silakan pilih program studi"))
        }

        if (_students.value.any { it.id != id && it.nim.equals(trimmedNim, ignoreCase = true) }) {
            return Result.failure(IllegalArgumentException("NIM $trimmedNim sudah terdaftar pada mahasiswa lain"))
        }

        _students.value = _students.value.map { student ->
            if (student.id == id) {
                student.copy(
                    nim = trimmedNim,
                    name = trimmedName,
                    programStudi = programStudi,
                )
            } else {
                student
            }
        }
        return Result.success(Unit)
    }

    fun deleteStudent(id: String) {
        _students.value = _students.value.filterNot { it.id == id }
    }

    fun getStudentById(id: String): Student? {
        return _students.value.find { it.id == id }
    }

    fun resetToSampleData() {
        _searchQuery.value = ""
        _students.value = initialStudents
    }
}

import com.example.quiz1.viewmodel.StudentViewModel

@Composable
fun StudentManagerApp(
    viewModel: StudentViewModel = viewModel(),
) {
    val navController = rememberNavController()

    NavHost(
        navController = navController,
        startDestination = "splash",
    ) {
        composable("splash") {
            SplashScreen {
                navController.navigate("home") {
                    popUpTo("splash") { inclusive = true }
                }
            }
        }
        composable("home") {
            HomeScreen(
                viewModel = viewModel,
                onNavigateToAdd = { navController.navigate("add_student") },
                onNavigateToEdit = { studentId ->
                    navController.navigate("edit_student/$studentId")
                },
            )
        }
        composable("add_student") {
            AddStudentScreen(
                viewModel = viewModel,
                onNavigateBack = { navController.popBackStack() },
            )
        }
        composable(
            route = "edit_student/{studentId}",
            arguments = listOf(navArgument("studentId") { type = NavType.StringType }),
        ) { backStackEntry ->
            val studentId = backStackEntry.arguments?.getString("studentId") ?: ""
            EditStudentScreen(
                studentId = studentId,
                viewModel = viewModel,
                onNavigateBack = { navController.popBackStack() },
            )
        }
    }
